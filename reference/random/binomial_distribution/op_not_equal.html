#include <iostream>
#include <random>

int main()
{
  std::binomial_distribution<> a(3, 0.5);
  std::binomial_distribution<> b(3, 0.7);

  if (a != b) {
    std::cout << "not equal" << std::endl;
  }
  else {
    std::cout << "equal" << std::endl;
  }
}
